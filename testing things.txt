// set position target and upward
// where is camera
m_v3Position = a_v3Position;
// what is it looking at
m_v3Target = a_v3Target;

m_v3Forward = glm::normalize(m_v3Target - m_v3Position);
// up line
m_v3Upward = glm::normalize(a_v3Upward);

// crossproduct for sideline
m_v3Rightward = glm::normalize(glm::cross(m_v3Forward, m_v3Upward));
// above camera position
m_v3Above = (m_v3Position + m_v3Upward);

m_m4View = glm::lookAt(m_v3Position, m_v3Target, m_v3Upward);

// calculate view

// clamp y axis so there isnt full rotation
float fLimit = glm::radians(89.0f);
m_v3PitchYawRoll.x = glm::clamp(m_v3PitchYawRoll.x, -fLimit, fLimit);

quaternion qYaw = glm::angleAxis(m_v3PitchYawRoll.y, vector3(0.0f, 1.0f, 0.0f));
quaternion qPitch = glm::angleAxis(m_v3PitchYawRoll.x, vector3(1.0f, 0.0f, 0.0f));
// must yaw than pitch
quaternion qOrientation = glm::cross(qPitch, qYaw);

// apply rotation to matrix
m_v3Forward = qOrientation * m_v3Rightward;
m_v3Upward = qOrientation * m_v3Upward;
m_v3Rightward = qOrientation * vector3(1.0f, 0.0f, -1.0f);

// positional vectors follow from the directional ones
m_v3Target = m_v3Position + m_v3Forward;
m_v3Above = m_v3Position + m_v3Upward;

m_m4View = glm::lookAt(m_v3Position, m_v3Target, m_v3Upward);


// in rigidbody globalize
//TODO: Globalize MinL and MaxL and store them in MinG and MaxG
m_v3MinG = vector3(m_m4ToWorld * vector4(m_v3MinL, 1.0f));
m_v3MaxG = vector3(m_m4ToWorld * vector4(m_v3MaxL, 1.0f));
//Globalize the center
m_v3CenterG = m_m4ToWorld * vector4(m_v3Center, 1.0f);

//Calculate the 8 corners of the oriented box
	std::vector<vector3> cornerList;

	cornerList.push_back(vector3(m_v3MinL.x, m_v3MinL.y, m_v3MinL.z)); //000
	cornerList.push_back(vector3(m_v3MinL.x, m_v3MinL.y, m_v3MaxL.z)); //001
	cornerList.push_back(vector3(m_v3MinL.x, m_v3MaxL.y, m_v3MinL.z)); //010
	cornerList.push_back(vector3(m_v3MinL.x, m_v3MaxL.y, m_v3MaxL.z)); //011

	cornerList.push_back(vector3(m_v3MaxL.x, m_v3MinL.y, m_v3MinL.z)); //100
	cornerList.push_back(vector3(m_v3MaxL.x, m_v3MinL.y, m_v3MaxL.z)); //101
	cornerList.push_back(vector3(m_v3MaxL.x, m_v3MaxL.y, m_v3MinL.z)); //110
	cornerList.push_back(vector3(m_v3MaxL.x, m_v3MaxL.y, m_v3MaxL.z)); //111

	//translate them to global space
	for (uint i = 0; i < 8; i++) {
		cornerList[i] = m_m4ToWorld * vector4(cornerList[i], 1.0f);
	}

	//from those points calculate a box that surround them
	RigidBody pTemp(cornerList);
	this->m_v3MaxG = pTemp.m_v3MaxL;
	this->m_v3MinG = pTemp.m_v3MinL;
